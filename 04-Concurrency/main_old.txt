package main_old

import (
	"fmt"
)

func wait_and_print(msg string, doneChan chan bool) {
	// time.Sleep(2 * time.Second)
	fmt.Println(msg)
	doneChan <- true
}

func print(msg string, doneChan chan bool) {
	fmt.Printf("This is the message %v", msg)
	doneChan <- true
	close(doneChan)
}
